set(FROZEN_UI "${CMAKE_CURRENT_SOURCE_DIR}/../../native/host/source/src/ui")

file(SHA256 "${FROZEN_UI}/VisualStyle.h" STYLE_H_HASH)
if(NOT STYLE_CPP_HASH STREQUAL "b1d5a0662916b3899245e72ab65f5b8c0c1d8c2b00df1335fa0493f9dc3f56e8" OR NOT STYLE_H_HASH STREQUAL "abcdf4c74f4a53619c86a4dea4a573a5efe3381fe76d5f817fdf8b0608ec8891")
  message(FATAL_ERROR "Frozen production VisualStyle source mismatch")
endif()
add_executable(style-cost-probe probe.cpp "${FROZEN_UI}/VisualStyle.cpp")
target_include_directories(style-cost-probe PRIVATE "${FROZEN_UI}")
target_link_libraries(style-cost-probe PRIVATE Qt6::Widgets)
