cmake_minimum_required(VERSION 3.24)
project(qt_queue_sanitizer_probe LANGUAGES CXX)
find_package(Qt6 6.4.2 REQUIRED COMPONENTS Core DBus)
add_executable(qt_queue qt_queue.cpp)
target_compile_features(qt_queue PRIVATE cxx_std_20)
target_compile_options(qt_queue PRIVATE -g -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(qt_queue PRIVATE -fsanitize=thread)
target_link_libraries(qt_queue PRIVATE Qt6::Core)
add_executable(known_race known_race.cpp)
target_compile_features(known_race PRIVATE cxx_std_20)
target_compile_options(known_race PRIVATE -g -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(known_race PRIVATE -fsanitize=thread)

add_executable(qt_dbus qt_dbus.cpp)
target_compile_features(qt_dbus PRIVATE cxx_std_20)
target_compile_options(qt_dbus PRIVATE -g -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(qt_dbus PRIVATE -fsanitize=thread)
target_link_libraries(qt_dbus PRIVATE Qt6::Core Qt6::DBus)

add_executable(known_deadlock known_deadlock.cpp)
target_compile_features(known_deadlock PRIVATE cxx_std_20)
target_compile_options(known_deadlock PRIVATE -g -fsanitize=thread -fno-omit-frame-pointer)
target_link_options(known_deadlock PRIVATE -fsanitize=thread)
