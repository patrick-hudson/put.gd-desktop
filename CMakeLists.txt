cmake_minimum_required(VERSION 3.24)
project(putgd VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)
include(cmake/Dependencies.cmake)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

set(PUTGD_SANITIZER "" CACHE STRING "Sanitizer: address-undefined or thread")
add_library(putgd_build_options INTERFACE)
target_compile_features(putgd_build_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(putgd_build_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(putgd_build_options INTERFACE -Wall -Wextra -Wpedantic)
  if(PUTGD_SANITIZER STREQUAL "address-undefined")
    target_compile_options(putgd_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(putgd_build_options INTERFACE -fsanitize=address,undefined)
  elseif(PUTGD_SANITIZER STREQUAL "thread")
    target_compile_options(putgd_build_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(putgd_build_options INTERFACE -fsanitize=thread)
  elseif(NOT PUTGD_SANITIZER STREQUAL "")
    message(FATAL_ERROR "Unknown PUTGD_SANITIZER: ${PUTGD_SANITIZER}")
  endif()
endif()

add_library(putgd_s3_runtime STATIC src/storage/s3/SdkRuntime.cpp)
target_include_directories(putgd_s3_runtime PUBLIC src)
target_link_libraries(putgd_s3_runtime PRIVATE ${AWSSDK_LINK_LIBRARIES} PUBLIC putgd_build_options)

add_library(putgd_core STATIC src/core/Settings.cpp src/core/SettingsService.cpp
  src/core/ClipboardPayload.cpp src/core/ArtifactStore.cpp src/core/LocalArchive.cpp
  src/core/UploadNaming.cpp src/core/StorageReceipt.cpp src/core/StorageProvider.cpp
  src/core/Operation.cpp src/core/HistoryStore.cpp)
target_include_directories(putgd_core PUBLIC src)
target_link_libraries(putgd_core PUBLIC putgd_build_options Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql)

add_library(putgd_app STATIC src/app/OperationService.cpp src/app/HistoryService.cpp src/app/LocalServer.cpp)
target_link_libraries(putgd_app PUBLIC putgd_core)
add_executable(putgd-cli src/cli/main.cpp)
target_link_libraries(putgd-cli PRIVATE putgd_app)
install(TARGETS putgd-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
add_executable(putgd src/main.cpp)
target_link_libraries(putgd PRIVATE putgd_build_options putgd_s3_runtime putgd_app Qt6::Widgets)
install(TARGETS putgd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(BUILD_TESTING)
  find_package(Qt6 6.4.2 REQUIRED COMPONENTS Test)
  add_executable(test_build_contract tests/unit/test_build_contract.cpp)
  target_link_libraries(test_build_contract PRIVATE putgd_build_options putgd_s3_runtime
    Qt6::Test Qt6::Widgets Qt6::Sql qt6keychain)
  add_test(NAME build_contract COMMAND test_build_contract)
  set_tests_properties(build_contract PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(test_sdk_contract tests/storage/s3/test_sdk_contract.cpp)
  target_link_libraries(test_sdk_contract PRIVATE putgd_s3_runtime Qt6::Test ${AWSSDK_LINK_LIBRARIES})
  add_test(NAME sdk_contract COMMAND test_sdk_contract)
  set_tests_properties(sdk_contract PROPERTIES TIMEOUT 30)
  foreach(contract payload archive upload_naming history operations)
    add_executable(test_${contract} tests/unit/test_${contract}.cpp)
    target_link_libraries(test_${contract} PRIVATE putgd_app Qt6::Test)
    target_include_directories(test_${contract} PRIVATE tests)
    add_test(NAME ${contract}_contract COMMAND test_${contract})
    set_tests_properties(${contract}_contract PROPERTIES TIMEOUT 60)
  endforeach()
  foreach(contract storage_contract cli config_reload)
    add_executable(test_${contract} tests/integration/test_${contract}.cpp)
    target_link_libraries(test_${contract} PRIVATE putgd_app Qt6::Test)
    target_include_directories(test_${contract} PRIVATE tests)
    add_test(NAME ${contract}_integration COMMAND test_${contract})
    set_tests_properties(${contract}_integration PROPERTIES TIMEOUT 60)
  endforeach()
  target_compile_definitions(test_cli PRIVATE PUTGD_CLI_PATH="$<TARGET_FILE:putgd-cli>")
  add_dependencies(test_cli putgd-cli)
  add_executable(test_settings tests/unit/test_settings.cpp)
  target_include_directories(test_settings PRIVATE src)
  target_link_libraries(test_settings PRIVATE putgd_core Qt6::Test)
  add_test(NAME settings_contract COMMAND test_settings)
  set_tests_properties(settings_contract PROPERTIES TIMEOUT 30)
endif()
