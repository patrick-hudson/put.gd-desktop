cmake_minimum_required(VERSION 3.24)
project(putgd VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)
include(cmake/Dependencies.cmake)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

set(PUTGD_SANITIZER "" CACHE STRING "Sanitizer: address-undefined or thread")
add_library(putgd_build_options INTERFACE)
target_compile_features(putgd_build_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(putgd_build_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(putgd_build_options INTERFACE -Wall -Wextra -Wpedantic)
  if(PUTGD_SANITIZER STREQUAL "address-undefined")
    target_compile_options(putgd_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(putgd_build_options INTERFACE -fsanitize=address,undefined)
  elseif(PUTGD_SANITIZER STREQUAL "thread")
    target_compile_options(putgd_build_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(putgd_build_options INTERFACE -fsanitize=thread)
  elseif(NOT PUTGD_SANITIZER STREQUAL "")
    message(FATAL_ERROR "Unknown PUTGD_SANITIZER: ${PUTGD_SANITIZER}")
  endif()
endif()

add_library(putgd_s3_runtime STATIC src/storage/s3/SdkRuntime.cpp)
target_include_directories(putgd_s3_runtime PUBLIC src)
target_link_libraries(putgd_s3_runtime PRIVATE ${AWSSDK_LINK_LIBRARIES} PUBLIC putgd_build_options)
# The SDK does not export its HTTP backend compile flags. Probe the actual installed
# SDK Curl implementation; retain the default factory on non-Curl SDK builds.
find_package(CURL QUIET)
if(CURL_FOUND)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_LIBRARIES ${AWSSDK_LINK_LIBRARIES} CURL::libcurl)
  check_cxx_source_compiles("#include <aws/core/http/curl/CurlHttpClient.h>
int main() { Aws::Http::CurlHttpClient::InitGlobalState(); Aws::Http::CurlHttpClient::CleanupGlobalState(); }" PUTGD_SDK_CURL_TRANSPORT)
  unset(CMAKE_REQUIRED_LIBRARIES)
  if(PUTGD_SDK_CURL_TRANSPORT)
    target_compile_definitions(putgd_s3_runtime PRIVATE PUTGD_SDK_CURL_TRANSPORT=1)
    target_link_libraries(putgd_s3_runtime PRIVATE CURL::libcurl)
  endif()
endif()


add_library(putgd_core STATIC src/core/Settings.cpp src/core/SettingsService.cpp
  src/platform/Geometry.cpp
  src/core/ClipboardPayload.cpp src/core/ArtifactStore.cpp src/core/LocalArchive.cpp
  src/core/UploadNaming.cpp src/core/StorageReceipt.cpp src/core/StorageProvider.cpp
  src/core/Operation.cpp src/core/HistoryStore.cpp)
target_include_directories(putgd_core PUBLIC src)
target_link_libraries(putgd_core PUBLIC putgd_build_options Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql)

add_library(putgd_s3 STATIC src/storage/s3/S3Provider.cpp src/storage/s3/PublicUrl.cpp src/secrets/CredentialStore.cpp)
target_link_libraries(putgd_s3 PUBLIC putgd_core putgd_s3_runtime PRIVATE ${AWSSDK_LINK_LIBRARIES} qt6keychain)

add_library(putgd_app STATIC src/app/OperationService.cpp src/app/HistoryService.cpp src/app/LocalServer.cpp src/app/LoginStartup.cpp)
target_link_libraries(putgd_app PUBLIC putgd_core)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PUTGD_XCB REQUIRED IMPORTED_TARGET xcb xcb-randr xcb-keysyms xcb-xfixes)
  add_library(putgd_x11 STATIC src/platform/x11/X11Capture.cpp
    src/platform/x11/X11Clipboard.cpp src/platform/x11/X11Shortcuts.cpp
    src/platform/x11/X11SourceAdapter.cpp src/ui/RegionOverlay.cpp)
  target_link_libraries(putgd_x11 PUBLIC putgd_app Qt6::Widgets PRIVATE PkgConfig::PUTGD_XCB)
endif()
add_library(putgd_ui STATIC src/ui/SettingsWindow.cpp src/ui/ShortcutEditor.cpp
  src/ui/RecentOperations.cpp src/ui/HistoryWindow.cpp src/ui/HistoryModel.cpp src/ui/TrayController.cpp)
target_link_libraries(putgd_ui PUBLIC putgd_app putgd_s3 Qt6::Widgets)
add_executable(putgd-cli src/cli/main.cpp)
target_link_libraries(putgd-cli PRIVATE putgd_app)
install(TARGETS putgd-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
add_executable(putgd src/main.cpp)
target_link_libraries(putgd PRIVATE putgd_build_options putgd_s3_runtime putgd_s3 putgd_app putgd_ui Qt6::Widgets)
if(TARGET putgd_x11)
  target_link_libraries(putgd PRIVATE putgd_x11)
  target_compile_definitions(putgd PRIVATE PUTGD_NATIVE_X11)
endif()
install(TARGETS putgd RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
